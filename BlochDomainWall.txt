    ({\x-0.3*cos(\ang)}, {-0.3*sin(\ang)}) -- ({\x+0.3*cos(\ang)}, {0.3*sin(\ang)});

\foreach \x in {1.5,2.5,3.5}
  \draw[-{Stealth[scale=1.5]}, fered, thick] (\x,0.5) -- (\x,-0.5);

\draw[dashed, gray, thick] (1,-1.4) -- (1, 1.4);
